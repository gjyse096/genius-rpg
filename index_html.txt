        function triggerVibe(pattern) {
            if (!state.settings || !state.settings.vibe) return;
            try {
                if (navigator.vibrate) navigator.vibrate(pattern);
            } catch (e) {}
        }

        const Sound = {
            ctx: null,
            play(freq, type, duration, vol = 0.12, endFreq = null) {
                if (!state.settings || !state.settings.sound) return;
                try {
                    if (!this.ctx) {
                        const AudioCtx = window.AudioContext || window.webkitAudioContext;
                        if (AudioCtx) this.ctx = new AudioCtx();
                    }
                    if (!this.ctx) return;
                    if (this.ctx.state === 'suspended') this.ctx.resume();

                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = type;
                    osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
                    if (endFreq) osc.frequency.exponentialRampToValueAtTime(endFreq, this.ctx.currentTime + duration);

                    gain.gain.setValueAtTime(vol, this.ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);

                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start();
                    osc.stop(this.ctx.currentTime + duration);
                } catch (e) {}
            },
            click() { this.play(580, 'sine', 0.05, 0.08); },
            hit(crit) {
                triggerVibe(crit ? [35, 30, 50] : 25);
                if (crit) {
                    this.play(440, 'sawtooth', 0.14, 0.16, 980);
                    setTimeout(() => this.play(880, 'triangle', 0.16, 0.18), 60);
                } else {
                    this.play(300, 'triangle', 0.12, 0.14, 600);
                }
            },
            wrong() {
                triggerVibe([80, 40, 80]);
                this.play(160, 'sawtooth', 0.25, 0.16, 70);
            },
            bossLoot() {
                triggerVibe([40, 30, 40, 30, 90]);
                [523, 659, 784, 1046, 1318].forEach((f, idx) => {
                    setTimeout(() => this.play(f, 'triangle', 0.18, 0.16), idx * 70);
                });
            },
            levelup() {
                triggerVibe([50, 40, 100]);
                [523, 659, 784, 1046].forEach((f, idx) => {
                    setTimeout(() => this.play(f, 'triangle', 0.2, 0.15), idx * 85);
                });
            }
        };

        function loadAccountsRegistry() {
            try {
                const raw = localStorage.getItem(ACCOUNTS_INDEX_KEY);
                accountsList = raw ? JSON.parse(raw) : [];
            } catch (e) {
                accountsList = [];
            }

            try {
                const legacyRaw = localStorage.getItem(LEGACY_SAVE_KEY);
                if (legacyRaw && accountsList.length === 0) {
                    const legacyData = JSON.parse(legacyRaw);
                    const migId = 'acc_' + Date.now();
                    const migAcc = { id: migId, name: '기존모험가', pin: 'test1234', createdAt: getTodayStr() };
                    accountsList.push(migAcc);
                    localStorage.setItem(ACCOUNTS_INDEX_KEY, JSON.stringify(accountsList));
                    const merged = mergeWithDefaults(legacyData);
                    localStorage.setItem(getAccountStorageKey(migId), JSON.stringify(merged));
                    localStorage.setItem(getAccountStorageKey(migId) + '_backup', JSON.stringify(merged));
                    localStorage.setItem(ACTIVE_USER_KEY, migId);
                    localStorage.removeItem(LEGACY_SAVE_KEY);
                }
            } catch (e) {}
        }

        function loadAccountVault(accId) {
            const key = getAccountStorageKey(accId);
            const backupKey = key + '_backup';
            let loaded = null;
            try {
                const raw = localStorage.getItem(key);
                if (raw) loaded = JSON.parse(raw);
            } catch (e) {}
            if (!loaded) {
                try {
                    const bRaw = localStorage.getItem(backupKey);
                    if (bRaw) loaded = JSON.parse(bRaw);
                } catch (e) {}
            }
            state = loaded ? mergeWithDefaults(loaded) : JSON.parse(JSON.stringify(DEFAULT_USER_DATA));
            applyTheme(state.settings.theme);
        }

        function sanitizeNicknameInput(inputEl, isRegister = false) {
            const raw = inputEl.value;
            // 한글(완성형+자모+천지인 아래아 ㆍ/ᆢ), 영문, 숫자, 밑줄(_), 마침표(.)만 허용하고 띄어쓰기 및 기타 특수기호 즉시 제거
            const cleaned = raw.replace(/[^a-zA-Z0-9가-힣ㄱ-ㅎㅏ-ㅣ\u318D\u119E\u11A2_.]/g, '');
            if (raw !== cleaned) {
                inputEl.value = cleaned;
                showToast('⚠️ 띄어쓰기 및 특수기호(_, . 제외)는 입력할 수 없습니다.');
                if (isRegister) {
                    const nameStatus = document.getElementById('reg-name-status');
                    nameStatus.textContent = '⚠️ 띄어쓰기 및 특수기호(_, . 제외)는 자동 차단됩니다.';
                    nameStatus.className = 'mt-1.5 text-[11px] font-extrabold text-amber-400';
                }
            }
        }

        function isValidStrongPassword(pw, nickname = '') {
            const lenOk = pw.length >= 4 && pw.length <= 20;
            const hasAlpha = /[a-zA-Z]/.test(pw);
            const hasNum = /[0-9]/.test(pw);
            const mixOk = hasAlpha && hasNum;

            const lower = pw.toLowerCase();
            const lowerNick = nickname.trim().toLowerCase();

            // 1) 고유 문자 개수 검사 (11111wwwwww 처럼 2~3종류만 반복하는 경우 차단 -> 최소 4종류 이상)
            const uniqueCount = new Set(lower).size;
            const uniqOk = pw.length > 0 && uniqueCount >= 4;

            // 2) 단일 문자 과다 편중 검사 (한 문자가 전체 길이의 50% 이상 차지하면 차단)
            const freqMap = {};
            let maxFreq = 0;
            for (const ch of lower) {
                freqMap[ch] = (freqMap[ch] || 0) + 1;
                if (freqMap[ch] > maxFreq) maxFreq = freqMap[ch];
            }
            const dominantRatioOk = pw.length > 0 && (maxFreq / pw.length) < 0.5;

            // 3) 연속 반복 및 뻔한 키보드/숫자 패턴 차단
            const weakPatterns = [
                '1234', '2345', '3456', '4567', '5678', '6789', '7890',
                '0987', '9876', '8765', '7654', '6543', '5432', '4321',
                'abcd', 'bcde', 'qwer', 'asdf', 'zxcv', 'pass', 'admin', 'test', 'genius', '0000', '1111', 'wwww'
            ];
            const hasRepeatedChars = /(.)\1\1/.test(lower);
            const hasWeakSequence = weakPatterns.some(pat => lower.includes(pat));
            const containsNick = lowerNick.length >= 2 && lower.includes(lowerNick);
            const notEasy = pw.length > 0 && dominantRatioOk && !hasRepeatedChars && !hasWeakSequence && !containsNick;

            let reason = '';
            if (!lenOk) reason = '비밀번호 길이를 4~20자로 맞춰주세요.';
            else if (!mixOk) reason = '영문과 숫자를 모두 섞어야 합니다.';
            else if (!uniqOk) reason = '서로 다른 문자를 4종류 이상 섞어 주세요.';
            else if (!dominantRatioOk || hasRepeatedChars) reason = '같은 문자나 숫자를 너무 많이 반복할 수 없습니다.';
            else if (containsNick) reason = '닉네임이 포함된 비밀번호는 사용할 수 없습니다.';
            else if (hasWeakSequence) reason = '1234, qwer, abcd 같은 쉬운 패턴은 사용할 수 없습니다.';

            return {
                lenOk,
                mixOk,
                uniqOk,
                notEasy,
                valid: lenOk && mixOk && uniqOk && notEasy,
                reason
            };
        }

        }

        function submitLogin() {
            const name = document.getElementById('login-name-input').value.trim();
            const pin = document.getElementById('login-pin-input').value.trim();
            if (!name) return showToast('계정 닉네임을 입력해 주세요.');
            if (!pin) return showToast('비밀번호를 입력해 주세요.');

            const acc = accountsList.find(a => a.name.trim().toLowerCase() === name.toLowerCase());
            if (!acc) {
                Sound.wrong();
                return showToast('❌ 존재하지 않는 닉네임입니다. 새 계정을 만들어 보세요!');
            }
            if (acc.pin !== pin) {
                Sound.wrong();
                return showToast('❌ 비밀번호가 일치하지 않습니다!');
            }
            enterLobbyWithAccount(acc);
        }

            if (isDuplicateNickname(name)) {
                Sound.wrong();
                nameInput.classList.remove('screen-shake');
                void nameInput.offsetWidth;
                nameInput.classList.add('screen-shake');
                nameInput.focus();
                return showToast('❌ 이미 사용 중인 닉네임입니다! 다른 닉네임으로 입력해 주세요.');
            }

            const pwCheck = isValidStrongPassword(pin, name);
            if (!pwCheck.valid) {
                Sound.wrong();
                pinInput.value = '';
                onRegisterInputChange();
                pinInput.classList.remove('screen-shake');
                void pinInput.offsetWidth;
                pinInput.classList.add('screen-shake');
                pinInput.focus();
                return showToast(`⚠️ 비밀번호가 너무 쉽습니다! (${pwCheck.reason})`);
            }

            const newAcc = {
                id: 'acc_' + Date.now(),
                name,
                pin,
                createdAt: getTodayStr()
            };
            accountsList.push(newAcc);
            saveAccountsRegistry();

            currentAccount = newAcc;
            state = JSON.parse(JSON.stringify(DEFAULT_USER_DATA));
            saveState();

            nameInput.value = '';
            pinInput.value = '';
            onRegisterInputChange();
            Sound.levelup();
            enterLobbyWithAccount(newAcc);
            showToast(`✨ 환영합니다, ${name}님! 계정이 생성되었습니다.`);
        }

        function getTodayStr() {
            const d = new Date();
            return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
        }

        function getYesterdayStr() {
            const d = new Date();
            d.setDate(d.getDate() - 1);
            return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
        }

        function getEquippedPet() {
            return PETS.find(p => p.id === state.equippedPet) || PETS[0];
        }

        function getStats() {
            const tier = getCurrentTier();
            const pet = getEquippedPet();
            const baseAtk = 30 + (state.upgrades.atk - 1) * 8;
            return {
                atk: Math.round(baseAtk * tier.atkMult * (1 + pet.atkBonus)),
                maxHp: 100 + (state.upgrades.hp - 1) * 20,
                maxMp: 100,
                goldMult: Number(((1 + (state.upgrades.gold - 1) * 0.15) * tier.goldMult * (1 + pet.goldBonus)).toFixed(2)),
                critRate: 0.10 + (state.upgrades.crit - 1) * 0.05
            };
        }

        function renderRadarChart() {
            const cats = [
                { key: 'logic',    label: '수리논리', angle: -Math.PI / 2 },
                { key: 'geometry', label: '공간기하', angle: 0 },
                { key: 'science',  label: '과학탐구', angle: Math.PI / 2 },
                { key: 'fusion',   label: '영재융합', angle: Math.PI }
            ];
            const cx = 120, cy = 100, maxR = 62;
            const scores = cats.map(c => {
                const st = state.categoryStats[c.key] || { correct: 0, total: 0 };
                const rate = st.total > 0 ? Math.round((st.correct / st.total) * 100) : 0;
                return { ...c, rate, total: st.total, correct: st.correct };
            });

            const totalSolved = scores.reduce((acc, s) => acc + s.total, 0);
            document.getElementById('analysis-total-solved').textContent = `누적 ${totalSolved}문제 풀이`;

            let gridSvg = '';
            [0.33, 0.66, 1.0].forEach(scale => {
                const pts = cats.map(c => `${cx + Math.cos(c.angle) * maxR * scale},${cy + Math.sin(c.angle) * maxR * scale}`).join(' ');
                gridSvg += `<polygon points="${pts}" fill="none" stroke="rgba(148,163,184,0.35)" stroke-width="1" />`;
            });
            cats.forEach(c => {
                gridSvg += `<line x1="${cx}" y1="${cy}" x2="${cx + Math.cos(c.angle) * maxR}" y2="${cy + Math.sin(c.angle) * maxR}" stroke="rgba(148,163,184,0.3)" stroke-width="1" />`;
            });

            const dataPts = scores.map(s => {
                const r = Math.max(8, (s.rate / 100) * maxR);
                return `${cx + Math.cos(s.angle) * r},${cy + Math.sin(s.angle) * r}`;
            }).join(' ');

            const labelsSvg = scores.map(s => {
                const lx = cx + Math.cos(s.angle) * (maxR + 24);
                const ly = cy + Math.sin(s.angle) * (maxR + 18) + 4;
                return `<text x="${lx}" y="${ly}" text-anchor="middle" fill="currentColor" font-size="11" font-weight="800">${s.label}</text>`;
            }).join('');

            document.getElementById('radar-svg').innerHTML = `
                ${gridSvg}
                <polygon points="${dataPts}" fill="rgba(56, 189, 248, 0.35)" stroke="#38bdf8" stroke-width="2.5" />
                ${labelsSvg}
            `;

            document.getElementById('radar-bars').innerHTML = scores.map(s => `
                <div class="flex items-center justify-between text-xs sub-box px-3 py-1.5 rounded-xl border">
                    <span class="font-bold text-main">${s.label}</span>
                    <span class="text-cyan-500 font-extrabold">${s.rate}% (${s.correct}/${s.total})</span>
                </div>
            `).join('');

            const played = scores.filter(s => s.total > 0);
            const weakBox = document.getElementById('weak-area-box');
            if (played.length === 0) {
                weakBox.innerHTML = '던전에 출격하여 문제를 풀면 4대 영재 역량과 취약 영역을 실시간 진단해 드립니다!';
            } else {
                const weakest = [...played].sort((a, b) => a.rate - b.rate)[0];
                weakBox.innerHTML = `🔍 <b>영재 진단 리포트:</b> 현재 <b>[${weakest.label}]</b> 영역 정답률이 <b>${weakest.rate}%</b>로 가장 보완이 필요합니다. 해당 챕터 집중 수련을 추천합니다!`;
            }
        }

        function buyUpgrade(key) {
            const cost = state.upgrades[key] * 100;
            if (state.gold < cost) return showToast('코인이 부족합니다!');
            state.gold -= cost;
            state.upgrades[key]++;
            Sound.levelup();
            saveState();
            updateLobbyUI();
            showToast('연구 강화 성공! 영구 능력치가 상승했습니다.');
        }

        function summonPet() {
            if (state.gem < 15) return showToast('젬(💎)이 15개 필요합니다!');
            state.gem -= 15;
            const pool = PETS;
            const picked = pool[Math.floor(Math.random() * pool.length)];
            if (!state.ownedPets.includes(picked.id)) {
                state.ownedPets.push(picked.id);
                state.equippedPet = picked.id;
                Sound.bossLoot();
                showToast(`🎉 신규 파트너 [${picked.icon} ${picked.name}] 소환 및 장착!`);
            } else {

        function openCheckinModal() {
            Sound.click();
            const today = getTodayStr();
            const yesterday = getYesterdayStr();
            let effectiveStreak = state.checkin.streak;
            if (state.checkin.lastDate && state.checkin.lastDate !== today && state.checkin.lastDate !== yesterday) {
                effectiveStreak = 0;
            }
            const claimedToday = state.checkin.lastDate === today;
            const nextDayNum = claimedToday ? effectiveStreak : (effectiveStreak % 7) + 1;

            document.getElementById('checkin-grid').innerHTML = CHECKIN_REWARDS.map(r => {
                const isDone = claimedToday ? r.day <= effectiveStreak : r.day < nextDayNum;
                const isTodayTarget = !claimedToday && r.day === nextDayNum;
                return `
                    <div class="rounded-xl p-2 border text-center ${isTodayTarget ? 'bg-amber-500/20 border-amber-400 animate-pulse' : isDone ? 'bg-emerald-500/15 border-emerald-500/40' : 'sub-box'}">
                        <div class="text-[10px] font-black text-muted">${r.day}일차</div>
                        <div class="text-xs font-extrabold text-main my-1">${r.label}</div>
                        <div class="text-[10px] font-bold ${isDone ? 'text-emerald-500' : isTodayTarget ? 'text-amber-500' : 'text-muted'}">
                            ${isDone ? '✔ 완료' : isTodayTarget ? '오늘!' : '대기'}
                        </div>
                    </div>
                `;
            }).join('');

            const btn = document.getElementById('btn-claim-checkin');
            btn.disabled = claimedToday;
            btn.textContent = claimedToday ? '오늘 출석 보상 수령 완료 ✔' : `${nextDayNum}일차 출석 보상 받기!`;
            document.getElementById('modal-checkin').classList.remove('hidden');
        }

        function claimDailyCheckin() {
            const today = getTodayStr();
            if (state.checkin.lastDate === today) return;

            const yesterday = getYesterdayStr();
            const newStreak = (state.checkin.lastDate === yesterday) ? (state.checkin.streak % 7) + 1 : 1;
            const reward = CHECKIN_REWARDS[newStreak - 1];

            state.checkin.lastDate = today;
            state.checkin.streak = newStreak;
            state.gold += reward.gold;
            state.gem += reward.gem;
            gainTierXp(reward.xp);

            Sound.bossLoot();
            saveState();
            updateLobbyUI();
            openCheckinModal();
            showToast(`🎉 ${newStreak}일차 출석 완료! (${reward.label} + ${reward.xp} XP)`);
        }

        function useSkill(type) {
            if (battle.locked || battle.bossPhase2) return;
            if (type === 'erase') {
                if (battle.playerMp < 30) return;
                const wrongIndices = [0, 1, 2, 3].filter(i => i !== battle.currentCorrectIdx && !document.getElementById(`choice-${i}`).disabled);
                if (wrongIndices.length <= 1) return showToast('더 이상 소거할 오답이 없습니다!');
                battle.playerMp -= 30;
                const target = wrongIndices[Math.floor(Math.random() * wrongIndices.length)];
                const btn = document.getElementById(`choice-${target}`);
                btn.disabled = true;
                btn.classList.add('opacity-25', 'line-through');
                Sound.click();
                showToast('💡 오답 소거 발동!');
            } else if (type === 'power') {
                if (battle.playerMp < 45 || battle.powerBuff) return;
                battle.playerMp -= 45;
                battle.powerBuff = true;
                Sound.click();
                showToast('⚡ 집중 강타 준비! 다음 정답 2.2배 피해!');
            } else if (type === 'shield') {
                if (battle.playerMp < 40 || battle.shieldBuff) return;
                battle.playerMp -= 40;
                battle.shieldBuff = true;
                Sound.click();
                showToast('🛡️ 논리 방어막 전개!');
            }
            updateBattleHUD();
        }

        function recordCategoryStat(catKey, isCorrect) {
            if (!state.categoryStats[catKey]) state.categoryStats[catKey] = { correct: 0, total: 0 };
            state.categoryStats[catKey].total++;
            if (isCorrect) state.categoryStats[catKey].correct++;
        }

        function handleAnswer(selectedIdx) {
            if (battle.locked) return;
            battle.locked = true;
            stopSmartTimer();

            const qId = battle.queue[battle.index];
            const qObj = QUESTIONS.find(q => q.id === qId);
            const isCorrect = selectedIdx === battle.currentCorrectIdx;
            const stats = getStats();

            for (let i = 0; i < 4; i++) {
                const btn = document.getElementById(`choice-${i}`);
                btn.disabled = true;
                if (i === battle.currentCorrectIdx) btn.classList.add('correct');
                else if (i === selectedIdx && !isCorrect) btn.classList.add('wrong');
            }

            recordCategoryStat(qObj.cat, isCorrect);

            if (isCorrect) {
                battle.combo++;
                battle.correctCount++;
                const isFever = battle.combo >= 3;
                const critChance = stats.critRate + (battle.relics.includes('archimedes') ? 0.20 : 0);
                const isCrit = Math.random() < critChance;

                let dmg = stats.atk * (1 + battle.combo * 0.1);
                if (isFever) dmg *= 2;
                if (battle.powerBuff) dmg *= 2.2;
                if (isCrit) dmg *= 1.8;
                if (battle.relics.includes('pythagoras') && (qObj.cat === 'logic' || qObj.cat === 'geometry')) dmg *= 1.4;
                dmg = Math.round(dmg);

                const finalDmg = Math.max(dmg, battle.enemyHp);
                battle.enemyHp = Math.max(0, battle.enemyHp - finalDmg);

                const mpGain = 15 + (battle.relics.includes('boyle') ? 20 : 0);
                battle.playerMp = Math.min(battle.playerMaxMp, battle.playerMp + mpGain);

                const speedBonus = battle.timerLeft > 0 ? 1.25 : 1.0;
                const relicGoldMult = battle.relics.includes('alchemist') ? 1.35 : 1.0;
                const modeMult = (battle.mode === 'REVENGE' ? 2 : 1) * (battle.isBossStage ? 3 : 1);
                const goldGain = Math.round(25 * stats.goldMult * (isFever ? 2 : 1) * speedBonus * relicGoldMult * modeMult);
                const xpGain = Math.round((battle.isBossStage ? 45 : 20) * speedBonus);

                battle.earnedGold += goldGain;
                battle.earnedXp += xpGain;
                state.gold += goldGain;
                gainTierXp(xpGain);

                if (battle.mode === 'REVENGE') {
                    state.wrongNotes = state.wrongNotes.filter(id => id !== qId);
                    battle.earnedGem += 1;
                    state.gem += 1;
                }

                Sound.hit(isCrit);
                const mSvg = document.getElementById('monster-svg');
                mSvg.classList.remove('monster-hit');
                void mSvg.offsetWidth;
                mSvg.classList.add('monster-hit');
                spawnDamagePopup(isCrit ? `💥 CRITICAL -${finalDmg}!` : `⚔️ -${finalDmg}`, isCrit ? 'text-amber-300' : 'text-emerald-300');

                if (battle.isBossStage) {
                    const chObj = CHAPTERS.find(c => c.id === battle.mode);
                    const bossGem = chObj ? chObj.gem : 15;
                    battle.earnedGem += bossGem;
                    state.gem += bossGem;
                    triggerBossLootExplosion(goldGain, bossGem);
                    updateBattleHUD();
                    saveState();
                    setTimeout(() => startBossKeywordFinisher(qObj), 900);
                    return;
                }

                updateBattleHUD();
                saveState();
                setTimeout(() => openExplainModal(true, qObj.exp, speedBonus > 1 ? '⚡ 스마트 빠른 풀이 보너스 +25% 적용!' : ''), 550);

            } else {
                battle.combo = 0;
                if (!state.wrongNotes.includes(qId)) state.wrongNotes.push(qId);

                if (battle.shieldBuff) {
                    battle.shieldBuff = false;
                    Sound.click();
                    spawnDamagePopup('🛡️ 논리 방어막 발동!', 'text-cyan-300');
                } else {
                    const dmgTaken = battle.isBossStage ? 30 : 20;
                    battle.playerHp = Math.max(0, battle.playerHp - dmgTaken);
                    Sound.wrong();
                    const frame = document.getElementById('app-frame');
                    frame.classList.remove('screen-shake');
                    void frame.offsetWidth;
                    frame.classList.add('screen-shake');
                    spawnDamagePopup(`💔 -${dmgTaken} HP`, 'text-rose-400');
                }

                updateBattleHUD();
                saveState();
                setTimeout(() => openExplainModal(false, qObj.exp, ''), 550);
            }
        }

        function renderBossKeywordUI(bf, allWords, qObj) {
            const b1 = battle.kwSelected[0] ? `<span class="text-emerald-500 underline font-black">[ ${battle.kwSelected[0]} ]</span>` : `<span class="text-amber-500 font-black">[ ① 빈칸 ]</span>`;
            const b2 = battle.kwSelected[1] ? `<span class="text-emerald-500 underline font-black">[ ${battle.kwSelected[1]} ]</span>` : `<span class="text-amber-500 font-black">[ ② 빈칸 ]</span>`;
            const formatted = bf.sentence.replace('[ ① ]', b1).replace('[ ② ]', b2);
            document.getElementById('kw-sentence').innerHTML = formatted;

            document.getElementById('kw-chips').innerHTML = allWords.map(w => {
                const used = battle.kwSelected.includes(w);
                return `
                    <button onclick="selectBossKeyword('${w}')" ${used ? 'disabled' : ''} class="choice-btn py-2.5 px-3 rounded-xl text-xs font-extrabold text-center ${used ? 'opacity-30' : ''}">
                        ${w}
                    </button>
                `;
            }).join('');
        }

        function selectBossKeyword(word) {
            const chObj = CHAPTERS.find(c => c.id === battle.mode);
            const bf = chObj.bossFinisher;
            battle.kwSelected.push(word);
            Sound.click();

            const allWords = Array.from(document.querySelectorAll('#kw-chips button')).map(b => b.textContent.trim());
            const qObj = QUESTIONS.find(q => q.id === battle.queue[battle.index]);
            renderBossKeywordUI(bf, allWords, qObj);

            if (battle.kwSelected.length === 2) {
                const isPerfect = battle.kwSelected[0] === bf.blanks[0] && battle.kwSelected[1] === bf.blanks[1];
                if (isPerfect) {
                    battle.earnedGold += 100;
                    battle.earnedGem += 5;
                    state.gold += 100;
                    state.gem += 5;
                    gainTierXp(30);
                    Sound.bossLoot();
                    saveState();
                    setTimeout(() => {
                        openExplainModal(true, qObj.exp, '🌟 보스 핵심 원리 조합 퍼펙트! (+100G · +5💎 · +30 XP)');
                    }, 350);
                } else {
                    Sound.wrong();
                    showToast('원리 순서가 맞지 않습니다! 다시 조합해 보세요.');
                    battle.kwSelected = [];
                    setTimeout(() => renderBossKeywordUI(bf, allWords, qObj), 400);
                }
            }
        }

                openRelicDraftModal();
            } else {
                loadStage(false);
            }
        }

        /* ==========================================================================
           [13] 투명 오버레이 연습장 + 풀이 스냅샷 저장
           ========================================================================== */
        const sp = {
            canvas: null,
            ctx: null,
            drawing: false,
            color: '#fde047',
            size: 3.5,
            isEraser: false,
            hasStrokes: false,
            lastX: 0,
            lastY: 0
        };

        function initScratchpad() {
            sp.canvas = document.getElementById('scratchpad-canvas');
            sp.ctx = sp.canvas.getContext('2d');

            const getPos = (e) => {
                const rect = sp.canvas.getBoundingClientRect();
                const clientX = e.touches ? e.touches[0].clientX : e.clientX;
                const clientY = e.touches ? e.touches[0].clientY : e.clientY;
                return { x: clientX - rect.left, y: clientY - rect.top };
            };

            const startDraw = (e) => {
                if (e.cancelable) e.preventDefault();
                sp.drawing = true;
                const pos = getPos(e);
                sp.lastX = pos.x;
                sp.lastY = pos.y;
                sp.ctx.beginPath();
                sp.ctx.moveTo(pos.x, pos.y);
            };

            const moveDraw = (e) => {
                if (!sp.drawing) return;
                if (e.cancelable) e.preventDefault();
                const pos = getPos(e);
                sp.ctx.lineCap = 'round';
                sp.ctx.lineJoin = 'round';
                if (sp.isEraser) {
                    sp.ctx.globalCompositeOperation = 'destination-out';
                    sp.ctx.lineWidth = 24;
                } else {
                    sp.ctx.globalCompositeOperation = 'source-over';
                    sp.ctx.strokeStyle = sp.color;
                    sp.ctx.lineWidth = sp.size;
                }
                sp.ctx.lineTo(pos.x, pos.y);
                sp.ctx.stroke();
                sp.lastX = pos.x;
                sp.lastY = pos.y;
                if (!sp.isEraser) sp.hasStrokes = true;
            };

            const endDraw = () => {
                sp.drawing = false;
                sp.ctx.closePath();
                sp.ctx.globalCompositeOperation = 'source-over';
            };

            sp.canvas.addEventListener('mousedown', startDraw);
            sp.canvas.addEventListener('mousemove', moveDraw);
            window.addEventListener('mouseup', endDraw);
            sp.canvas.addEventListener('touchstart', startDraw, { passive: false });
            sp.canvas.addEventListener('touchmove', moveDraw, { passive: false });
            sp.canvas.addEventListener('touchend', endDraw);
        }

        function resizeScratchpadCanvas() {
            if (!sp.canvas) return;
            const container = document.getElementById('sp-canvas-container');
            const rect = container.getBoundingClientRect();
            const dpr = window.devicePixelRatio || 1;
            const temp = document.createElement('canvas');
            temp.width = sp.canvas.width;
            temp.height = sp.canvas.height;
            if (temp.width > 0 && temp.height > 0) temp.getContext('2d').drawImage(sp.canvas, 0, 0);

            sp.canvas.width = rect.width * dpr;
            sp.canvas.height = rect.height * dpr;
            sp.ctx.scale(dpr, dpr);
            if (temp.width > 0 && temp.height > 0) {
                sp.ctx.drawImage(temp, 0, 0, temp.width / dpr, temp.height / dpr);
            }
        }

        function openScratchpad() {
            Sound.click();
            document.getElementById('sp-question-mirror').textContent = document.getElementById('q-text').textContent;
            document.getElementById('scratchpad-overlay').classList.remove('hidden');
            requestAnimationFrame(() => resizeScratchpadCanvas());
        }

        function saveScratchpadSnapshot() {
            if (!sp.hasStrokes) return showToast('먼저 연습장에 풀이 과정을 적어주세요!');
            const qId = battle.queue[battle.index];
            const qObj = QUESTIONS.find(q => q.id === qId);

            const exportCanvas = document.createElement('canvas');
            exportCanvas.width = sp.canvas.width;
            exportCanvas.height = sp.canvas.height;
            const eCtx = exportCanvas.getContext('2d');
            eCtx.fillStyle = state.settings.theme === 'light' ? '#f8fafc' : '#0b1120';
            eCtx.fillRect(0, 0, exportCanvas.width, exportCanvas.height);
            eCtx.drawImage(sp.canvas, 0, 0);

            const dataUrl = exportCanvas.toDataURL('image/png', 0.7);
            state.savedNotes.unshift({
                tag: qObj.tag,
                q: qObj.q,
                date: getTodayStr(),
                img: dataUrl
            });
            if (state.savedNotes.length > 15) state.savedNotes.pop();
            saveState();
            Sound.levelup();
            showToast('📸 풀이가 [오답/노트 → 비밀 풀이노트]에 저장되었습니다!');
        }
